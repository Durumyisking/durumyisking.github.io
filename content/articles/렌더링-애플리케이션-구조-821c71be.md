인터럽트가 있을때는 application.Run()을 처리한다

Main.cpp 

```c++
BOOL InitInstance(HINSTANCE hInstance, int nCmdShow)
{
   hInst = hInstance; // 인스턴스 핸들을 전역 변수에 저장합니다.

   HWND hWnd = CreateWindowW(szWindowClass, szTitle, WS_OVERLAPPEDWINDOW,
       0, 0, 1600, 900, nullptr, nullptr, hInstance, nullptr);

   if (!hWnd)
   {
      return FALSE;
   }
   
   application.SetWindow(hWnd, 1600, 900);
   application.Initialize();  ------------ 애플리케이션 초기화
   ShowWindow(hWnd, nCmdShow);
   UpdateWindow(hWnd);

   return TRUE;
}


while (true)
    {
        if (PeekMessage(&msg, nullptr, 0, 0, PM_REMOVE))
        {
            if (WM_QUIT == msg.message)
                break;

            if (!TranslateAccelerator(msg.hwnd, hAccelTable, &msg))
            {
                TranslateMessage(&msg);
                DispatchMessage(&msg);
            }
        }
        else
        {
            application.Run(); // HighLevelInterface  
        }
    }
```

<summary>HighLevelInterface.Run()</summary>

Application.cpp

```c++
void CApplication::Run()
	{
		update();
		fixedupdate();
		render();
	}

void CApplication::update()
	{
		CTimeMgr::update();
		Input::update();
		CSceneMgr::update();
	}
	void CApplication::fixedupdate()
	{
		CSceneMgr::fixedupdate();
	}
	void CApplication::render()
	{
		CTimeMgr::Render(mHdc);

		graphicDevice->Clear();
		graphicDevice->AdjustViewPorts();

		CSceneMgr::render();

		graphicDevice->Present();
	}
```

Scene.cpp

```c++
void CScene::update()
	{
		for (CLayer& layer : mLayers)
		{
			layer.update(); // layer은 GameObj타입 벡터
		}
	}

	void CScene::fixedupdate()
	{
		for (CLayer& layer : mLayers)
		{
			layer.fixedupdate();
		}

	}

	void CScene::render()
	{
		for (CLayer& layer : mLayers)
		{
			layer.render();
		}
	}
```

Layer.cpp

```c++
void CLayer::update()
	{
		for (CGameObj* Obj : mGameObjs)
		{
			if (nullptr == Obj)
				continue;
			Obj->update(); // Layer벡터 내에 해당 타입의 GameObj가 있다
		}
	}

	void CLayer::fixedupdate()
	{
		for (CGameObj* Obj : mGameObjs)
		{
			if (nullptr == Obj)
				continue;
			Obj->fixedupdate();
		}
	}

	void CLayer::render()
	{
		for (CGameObj* Obj : mGameObjs)
		{
			if (nullptr == Obj)
				continue;
			Obj->render();
		}
	}
```

Gameobj.cpp

```c++
void CGameObj::update()
	{
		for (CComponent* comp : mComponents)
		{
			if (nullptr == comp)
				continue;			
			comp->update(); // Gameobj는 다양한 Component를 가짐 해당 Component들을 처리함
											// Component 클래스는 순수가상함수
		}
	}

	void CGameObj::fixedupdate()
	{
		for (CComponent* comp : mComponents)
		{
			if (nullptr == comp)
				continue;
			comp->fixedupdate();
		}
	}

	void CGameObj::render()
	{
		for (CComponent* comp : mComponents)
		{
			if (nullptr == comp)
				continue;
			comp->render();
		}
	}
```
